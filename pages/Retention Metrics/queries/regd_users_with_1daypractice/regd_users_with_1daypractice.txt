WITH sept_users AS (
  SELECT "id" AS "userId", "createdAt" AS user_created_at
  FROM "User"
  WHERE DATE_TRUNC('month', "createdAt") = '2025-09-01'
),
practice_days AS (
  SELECT "userId", COUNT(DISTINCT DATE("eventDate")) AS practice_day_count
  FROM "DailyUserEvent"
  WHERE "event" = 'Question'
  GROUP BY "userId"
)
SELECT u."userId", u.user_created_at
FROM sept_users u
JOIN practice_days p ON u."userId" = p."userId"
WHERE p.practice_day_count = 1;
