SELECT 
  'Active Users' as user_type,
  COUNT(DISTINCT p."userId") as user_count,
  SUM(p."amount") as total_revenue,
  ROUND(AVG(p."amount"), 2) as avg_revenue_per_user
FROM "Payment" p
WHERE p."status" = 'responseReceivedSuccess' 
  AND p."createdAt" > '2025-04-01'
  AND EXISTS (
		SELECT 1 FROM "DailyUserEvent" due 
		WHERE due."userId" = p."userId" 
		AND due."eventDate" >= CURRENT_DATE - INTERVAL '30 days'
	)

UNION ALL

SELECT 
  'Churned Users (30d)' as user_type,
  COUNT(DISTINCT p."userId") as user_count,
  SUM(p."amount") as total_revenue,
  ROUND(AVG(p."amount"), 2) as avg_revenue_per_user
FROM "Payment" p
WHERE p."status" = 'responseReceivedSuccess' 
  AND p."createdAt" > '2025-04-01'
  AND NOT EXISTS (
		SELECT 1 FROM "DailyUserEvent" due 
		WHERE due."userId" = p."userId" 
		AND due."eventDate" >= CURRENT_DATE - INTERVAL '30 days'
	);
